```sql
WITH cases AS (
  SELECT DISTINCT RefNo, CaseNo
  FROM `mitelman-db.prod.Cytogen`
  WHERE Morph = '3111' AND Topo = '0401'
),
mit_cc AS (
  SELECT DISTINCT cc.RefNo, cc.CaseNo, cc.Chr, cc.Start, cc.End, cc.Type
  FROM `mitelman-db.prod.CytoConverted` cc
  JOIN cases c ON c.RefNo = cc.RefNo AND c.CaseNo = cc.CaseNo
),
mit_total AS (
  SELECT COUNT(DISTINCT CONCAT(CAST(RefNo AS STRING), '_', CaseNo)) AS total FROM mit_cc
),
mitelman AS (
  SELECT
    b.chromosome,
    b.cytoband_name,
    100 * SAFE_DIVIDE(COUNT(DISTINCT IF(m.Type = 'Gain', CONCAT(CAST(m.RefNo AS STRING), '_', m.CaseNo), NULL)), ANY_VALUE(t.total)) AS freq_gain,
    100 * SAFE_DIVIDE(COUNT(DISTINCT IF(m.Type = 'Loss', CONCAT(CAST(m.RefNo AS STRING), '_', m.CaseNo), NULL)), ANY_VALUE(t.total)) AS freq_loss
  FROM `mitelman-db.prod.CytoBands_hg38` b
  CROSS JOIN mit_total t
  LEFT JOIN mit_cc m
    ON m.Chr = b.chromosome AND m.Start < b.hg38_stop AND m.End > b.hg38_start
  GROUP BY b.chromosome, b.cytoband_name
),
tcga_seg AS (
  SELECT case_barcode, chromosome, start_pos, end_pos, copy_number
  FROM `isb-cgc-bq.TCGA.copy_number_segment_allelic_hg38_gdc_current`
  WHERE project_short_name = 'TCGA-BRCA'
),
tcga_total AS (
  SELECT COUNT(DISTINCT case_barcode) AS total FROM tcga_seg
),
tcga AS (
  SELECT
    b.chromosome,
    b.cytoband_name,
    100 * SAFE_DIVIDE(COUNT(DISTINCT IF(s.copy_number >= 4, s.case_barcode, NULL)), ANY_VALUE(t.total)) AS freq_amp,
    100 * SAFE_DIVIDE(COUNT(DISTINCT IF(s.copy_number = 3, s.case_barcode, NULL)), ANY_VALUE(t.total)) AS freq_gain,
    100 * SAFE_DIVIDE(COUNT(DISTINCT IF(s.copy_number = 0, s.case_barcode, NULL)), ANY_VALUE(t.total)) AS freq_homodel,
    100 * SAFE_DIVIDE(COUNT(DISTINCT IF(s.copy_number = 1, s.case_barcode, NULL)), ANY_VALUE(t.total)) AS freq_heterodel
  FROM `mitelman-db.prod.CytoBands_hg38` b
  CROSS JOIN tcga_total t
  LEFT JOIN tcga_seg s
    ON s.chromosome = b.chromosome AND s.start_pos < b.hg38_stop AND s.end_pos > b.hg38_start
  GROUP BY b.chromosome, b.cytoband_name
),
paired AS (
  SELECT
    m.chromosome,
    p.aberration,
    p.mitelman_freq,
    p.tcga_freq
  FROM mitelman m
  JOIN tcga g ON g.chromosome = m.chromosome AND g.cytoband_name = m.cytoband_name,
  UNNEST([
    STRUCT('amplification' AS aberration, m.freq_gain AS mitelman_freq, g.freq_amp AS tcga_freq),
    STRUCT('gain', m.freq_gain, g.freq_gain),
    STRUCT('homozygous_deletion', m.freq_loss, g.freq_homodel),
    STRUCT('heterozygous_deletion', m.freq_loss, g.freq_heterodel)
  ]) AS p
)
SELECT
  chromosome,
  aberration,
  COUNT(*) AS n,
  CORR(mitelman_freq, tcga_freq) AS correlation,
  `isb-cgc-bq.functions.corr_pvalue_current`(CORR(mitelman_freq, tcga_freq), COUNT(*)) AS p_value
FROM paired
GROUP BY chromosome, aberration
HAVING COUNT(*) >= 5
ORDER BY chromosome, aberration
```
