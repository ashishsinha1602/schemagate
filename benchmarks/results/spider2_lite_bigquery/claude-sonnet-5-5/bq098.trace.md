WITH t AS (
  SELECT
    z.borough AS pickup_borough,
    IF(y.total_amount = 0, 0, (y.tip_amount * 100) / y.total_amount) AS tip_rate
  FROM `bigquery-public-data.new_york_taxi_trips.tlc_yellow_trips_*` y
  JOIN `bigquery-public-data.new_york_taxi_trips.taxi_zone_geom` z
    ON y.pickup_location_id = z.zone_id
  WHERE y._TABLE_SUFFIX = '2016'
    AND y.pickup_datetime >= TIMESTAMP('2016-01-01 00:00:00')
    AND y.pickup_datetime < TIMESTAMP('2016-01-08 00:00:00')
    AND y.dropoff_datetime >= TIMESTAMP('2016-01-01 00:00:00')
    AND y.dropoff_datetime < TIMESTAMP('2016-01-08 00:00:00')
    AND y.dropoff_datetime > y.pickup_datetime
    AND y.passenger_count > 0
    AND y.trip_distance >= 0
    AND y.tip_amount >= 0
    AND y.tolls_amount >= 0
    AND y.mta_tax >= 0
    AND y.fare_amount >= 0
    AND y.total_amount >= 0
)
SELECT
  pickup_borough,
  100 * COUNTIF(tip_rate = 0) / COUNT(*) AS no_tip_percentage
FROM t
GROUP BY pickup_borough
