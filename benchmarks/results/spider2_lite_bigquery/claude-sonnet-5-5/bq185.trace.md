```sql
WITH trips from the 2016 yellow table, joined to taxi_zone_geom twice (pickup and dropoff) and filtered to Brooklyn. Pickup dates run from Feb 1 to Feb 7 inclusive, so the filter is `pickup_datetime >= '2016-02-01'` and `< '2016-02-08'`. "More than three passengers" means `passenger_count > 3`, and "at least ten miles" means `trip_distance >= 10`. The duration is positive when `dropoff_datetime > pickup_datetime`.

SELECT AVG(TIMESTAMP_DIFF(t.dropoff_datetime, t.pickup_datetime, SECOND) / 60.0) AS avg_trip_duration_minutes
FROM `bigquery-public-data.new_york_taxi_trips.tlc_yellow_trips_*` t
JOIN `bigquery-public-data.new_york_taxi_trips.taxi_zone_geom` pz
  ON t.pickup_location_id = pz.zone_id
JOIN `bigquery-public-data.new_york_taxi_trips.taxi_zone_geom` dz
  ON t.dropoff_location_id = dz.zone_id
WHERE t._TABLE_SUFFIX = '2016'
  AND t.pickup_datetime >= TIMESTAMP('2016-02-01')
  AND t.pickup_datetime < TIMESTAMP('2016-02-08')
  AND t.dropoff_datetime > t.pickup_datetime
  AND t.passenger_count > 3
  AND t.trip_distance >= 10
  AND pz.borough = 'Brooklyn'
  AND dz.borough = 'Brooklyn'
```
