## Question
Use CPTAC proteomics and RNAseq data for Clear Cell Renal Cell Carcinoma to select 'Primary Tumor' and 'Solid Tissue Normal' samples. Join the datasets on sample submitter IDs and gene symbols. Calculate the correlation between protein abundance (log2 ratio) and gene expression levels (log-transformed+1 FPKM) for each gene and sample type. Filter out correlations with an absolute value greater than 0.5, and compute the average correlation for each sample type.

WITH prot AS (
  SELECT DISTINCT m.sample_submitter_id, p.gene_symbol, p.protein_abundance_log2ratio, p.aliquot_id
  FROM `isb-cgc-bq.CPTAC.quant_proteome_CPTAC_CCRCC_discovery_study_pdc_current` p
  JOIN `isb-cgc-bq.PDC_metadata.aliquot_to_case_mapping_current` m
    ON p.aliquot_id = m.aliquot_id
),
joined AS (
  SELECT
    r.sample_type_name AS sample_type,
    p.gene_symbol,
    p.protein_abundance_log2ratio AS protein_abundance,
    LOG10(r.fpkm_unstranded + 1) AS log_fpkm
  FROM prot p
  JOIN `isb-cgc-bq.CPTAC.RNAseq_hg38_gdc_current` r
    ON p.sample_submitter_id = r.sample_barcode
   AND p.gene_symbol = r.gene_name
  WHERE r.sample_type_name IN ('Primary Tumor', 'Solid Tissue Normal')
),
gene_corr AS (
  SELECT sample_type, gene_symbol, CORR(protein_abundance, log_fpkm) AS correlation
  FROM joined
  GROUP BY sample_type, gene_symbol
)
SELECT sample_type, AVG(correlation) AS avg_correlation
FROM gene_corr
WHERE correlation IS NOT NULL AND ABS(correlation) > 0.5
GROUP BY sample_type
