```sql
WITH pt AS (
  SELECT DISTINCT plot_inventory_year, plot_state_code, plot_state_code_name, plot_county_code, plot_phase_2_plot_number
  FROM `bigquery-public-data.usfs_fia.plot_tree`
  WHERE plot_inventory_year IN (2015, 2016, 2017)
),
base AS (
  SELECT
    pt.plot_inventory_year AS year,
    pt.plot_state_code_name AS state,
    IF(p.adjustment_factor_for_the_subplot > 0,
       p.area_used_to_calculate_all_expansion_factors * c.condition_proportion_unadjusted * p.adjustment_factor_for_the_subplot, 0) AS subplot_size,
    IF(p.adjustment_factor_for_the_macroplot > 0,
       p.area_used_to_calculate_all_expansion_factors * c.condition_proportion_unadjusted * p.adjustment_factor_for_the_macroplot, 0) AS macroplot_size
  FROM `bigquery-public-data.usfs_fia.condition` c
  JOIN pt
    ON c.state_code = pt.plot_state_code
   AND c.county_code = pt.plot_county_code
  JOIN `bigquery-public-data.usfs_fia.population` p
    ON p.state_code = pt.plot_state_code
   AND p.county_code = pt.plot_county_code
   AND p.phase_2_plot_number = pt.plot_phase_2_plot_number
  WHERE p.evaluation_type = 'EXPCURR'
    AND c.condition_status_code = 1
),
agg AS (
  SELECT year, state, AVG(subplot_size) AS avg_subplot_size, AVG(macroplot_size) AS avg_macroplot_size
  FROM base
  GROUP BY year, state
),
sub AS (
  SELECT 'subplot' AS plot_type, year, state, avg_subplot_size AS avg_size
  FROM agg
  QUALIFY ROW_NUMBER() OVER (PARTITION BY year ORDER BY avg_subplot_size DESC) = 1
),
mac AS (
  SELECT 'macroplot' AS plot_type, year, state, avg_macroplot_size AS avg_size
  FROM agg
  QUALIFY ROW_NUMBER() OVER (PARTITION BY year ORDER BY avg_macroplot_size DESC) = 1
)
SELECT plot_type, year, state, avg_size FROM sub
UNION ALL
SELECT plot_type, year, state, avg_size FROM mac
ORDER BY plot_type, year
```
